# TFE: the West's burdens in 395. Two reforms WRE starts with (10_countries) and cannot drop: each brings its own slot,
# so the crown keeps its usual free ones. The Reform of the Coinage ends the Debased Coinage (auto_modifiers).

# The clarissimi: senators excused the sordid munera, their land-tax arrears remitted, their estates beyond the census
tfe_senatorial_immunities = {
	potential = {
		has_or_had_tag = WRE
	}
	locked = {
		always = yes	# a way to lift the burdens is still to be designed
	}
	country_modifier = {
		global_nobles_estate_power = 0.5
		nobles_estate_target_satisfaction = medium_privilege_target_satisfaction
		nobles_estate_max_tax = -0.25
		government_reform_slots = 1
	}
}

# Patrocinium: coloni shelter under a great landlord, who hides them from the recruiting officer and pays gold instead
tfe_patrocinium = {
	potential = {
		has_or_had_tag = WRE
	}
	locked = {
		always = yes	# a way to lift the burdens is still to be designed
	}
	country_modifier = {
		global_peasants_estate_power = 0.25
		global_nobles_estate_power = 0.25	# the patron is a great landlord
		peasants_estate_target_satisfaction = medium_privilege_target_satisfaction
		peasants_estate_levy_size = -0.5
		global_manpower_modifier = -0.25
		government_reform_slots = 1
	}
}

# A sound coin again: the debased nummus called in and the soldiers paid in full-weight metal
tfe_coinage_reform = {
	potential = {
		has_or_had_tag = WRE
		NOT = { has_variable = tfe_coinage_restored }
	}
	allow = {
		at_war = no
		stability >= 25
	}
	country_modifier = {
		tax_income_efficiency = -0.1
	}
	on_fully_activated = {
		set_variable = { name = tfe_coinage_restored }	# ends the Debased Coinage
	}
	years = 3
}

# TFE: burdens the Roman state laid on its own people, locked like the West's others (tfe_late_roman_west.txt):
# each brings its own slot. The West's go when it gives up the law holding each (script/roman_burdens.py); the East's
# Disarmed Demos stays.

# The plebs are forbidden arms (Theodosian Code, on private weapons), so the state fields no levy and pays only professionals
tfe_disarmed_plebs = {
	potential = {
		OR = { has_or_had_tag = WRE has_variable = tfe_western_rome }
	}
	locked = {
		always = yes
	}
	country_modifier = {
		global_levy_size_modifier = -5	# not -1: advances, estates and laws stacked levies back on top of it
		army_maintenance_efficiency = 0.5
		peasants_estate_target_satisfaction = -0.1
		government_reform_slots = 1
	}
}

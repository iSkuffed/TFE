# TFE: the Hunnic Yoke, the Huns' tribute empire (vanilla tatar_yoke without its tax collector). Seeded at start
# (setup/395/15_international_organizations.txt). Nobody joins or leaves by diplomacy: war does it
# (casus_belli/tfe_hunnic_storm.txt), and the Hunnic Storm situation (situations/tfe_hunnic_storm.txt) may break it.
tfe_hunnic_yoke = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no

	has_leader_country = yes
	leader_title_key = "tfe_hunnic_yoke_LEADER"
	leader_type = character
	leader = {
		leader_country ?= {
			ruler ?= {
				add_to_list = leaders
			}
		}
	}
	leader_change_trigger_type = none

	diplomatic_capacity_cost = {
		add = {
			desc = "UPKEEP_BASE_VALUE"
			value = 1
		}
	}

	can_declare_war = {
		always = yes
	}

	# the tribute is the members' whole burden
	leader_modifier = {
		monthly_prestige = 0.1
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = yes
	}

	can_lead_trigger = {
		has_ruler = yes
		ruler ?= {
			is_adult = yes
		}
	}

	can_join_trigger = {
		always = no
	}

	can_leave_trigger = {
		always = no
	}

	auto_disband_trigger = {
		OR = {
			NOT = { country_exists = c:HNS }
			NOT = {
				country_has_special_status = {
					type = special_status:tfe_hunnic_overlord
					country = c:HNS
				}
			}
		}
	}

	international_organization_modifier = {
		cannot_declare_no_cb_wars_on_members = yes
	}

	payments_implemented = {
		tfe_hunnic_tribute
	}

	special_statuses_implemented = {
		tfe_hunnic_overlord
	}
}

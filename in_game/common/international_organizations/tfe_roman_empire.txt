# honorius-only: the Imperium Romanum's seat and Unity (tools/test_western_rome.py)
# TFE: the Imperium Romanum - one empire under two Augusti (395). Unity decides how far the halves act as one.
tfe_roman_empire = {
	has_target = no
	unique = yes
	show_on_diplomatic_map = yes
	member_color = rgb { 120 30 100 }

	# the Senior Augustus is the leader country and decrees the edicts (special_statuses/tfe_roman_empire.txt)
	has_leader_country = yes
	# the two Augusti at the top of the window, West then East (the members' order)
	leader_type = character
	leader_change_trigger_type = none
	leader_change_method = none
	leader_title_key = "tfe_roman_empire_LEADER"
	disband_if_no_leader = no
	leader = {
		every_international_organization_member = {
			ruler_or_heir_if_regent ?= { add_to_list = leaders }	# the boy Augustus, not his regent
		}
	}

	special_statuses_implemented = { tfe_senior_augustus }

	show_as_overlord_on_map_trigger = { always = yes }
	create_visible_trigger = { always = no }
	invite_visible_trigger = { always = no }

	variables = {
		tfe_unity = {
			format = "tfe_unity_DISPLAY"
			change_format = "VARIABLE_CHANGE_FORMAT"
			start = 65
			min = 0
			max = 100
			monthly_change = {
				add = {
					if = {
						limit = { c:WRE ?= { ruler_or_heir_if_regent ?= { dynasty ?= c:EAR.ruler_or_heir_if_regent.dynasty } } }
						add = { desc = "TFE_UNITY_ONE_DYNASTY" value = 0.1 }
					}
					if = {
						limit = { c:EAR ?= { any_country_at_war_with = { is_at_war_with = c:WRE } } }
						add = { desc = "TFE_UNITY_COMMON_ENEMY" value = 0.15 }
					}
				}
				# No blind drift: every loss has a cause the player can see and act on
				subtract = {
					# Latin West, Greek East: the courts no longer speak one tongue
					if = {
						limit = { c:WRE ?= { NOT = { court_language = c:EAR.court_language } } }
						add = { desc = "TFE_UNITY_TWO_TONGUES" value = 0.1 }
					}
					# A boy Augustus: the regent's court, not the emperor's, faces the other half (Stilicho in 395)
					if = {
						limit = { c:WRE ?= { has_regent = yes } }
						add = { desc = "TFE_UNITY_WEST_REGENCY" value = 0.1 }
					}
					if = {
						limit = { c:EAR ?= { has_regent = yes } }
						add = { desc = "TFE_UNITY_EAST_REGENCY" value = 0.1 }
					}
					if = {
						limit = {
							OR = {
								c:WRE ?= { is_rival_of = c:EAR }
								c:EAR ?= { is_rival_of = c:WRE }
							}
						}
						add = { desc = "TFE_UNITY_RIVALS" value = 0.2 }
					}
					if = {
						limit = { c:WRE ?= { NOT = { religion = c:EAR.religion } } }
						add = { desc = "TFE_UNITY_CREEDS" value = 0.2 }
					}
					if = {
						limit = { c:WRE ?= { opinion = { target = c:EAR value < 0 } } }
						add = { desc = "TFE_UNITY_WEST_RESENTS" value = 0.1 }
					}
					if = {
						limit = { c:EAR ?= { opinion = { target = c:WRE value < 0 } } }
						add = { desc = "TFE_UNITY_EAST_RESENTS" value = 0.1 }
					}
					if = {
						limit = {
							OR = {
								c:WRE ?= { any_core_location = { owner = c:EAR } }
								c:EAR ?= { any_core_location = { owner = c:WRE } }
							}
						}
						add = { desc = "TFE_UNITY_DISPUTED_LANDS" value = 0.1 }
					}
					# Closing the roads to the other Augustus; the IO grants access itself from 75 up
					if = {
						limit = { var:tfe_unity < 75 c:WRE ?= { NOT = { gives_military_access_to = c:EAR } } }
						add = { desc = "TFE_UNITY_ACCESS_DENIED" value = 0.1 }
					}
					if = {
						limit = { var:tfe_unity < 75 c:EAR ?= { NOT = { gives_military_access_to = c:WRE } } }
						add = { desc = "TFE_UNITY_ACCESS_DENIED" value = 0.1 }
					}
					if = {
						limit = { c:WRE ?= { is_at_war_with = c:EAR } }
						add = { desc = "TFE_UNITY_CIVIL_WAR" value = 1 }
					}
					# Gold for the Huns: a half buys peace for itself alone (situations/tfe_hunnic_storm.txt)
					if = {
						limit = {
							OR = {
								c:WRE ?= { is_member_of_international_organization = international_organization:tfe_hunnic_yoke }
								c:EAR ?= { is_member_of_international_organization = international_organization:tfe_hunnic_yoke }
							}
						}
						add = { desc = "TFE_UNITY_HUNNIC_SUBSIDY" value = 0.1 }
					}
				}
			}
		}
	}

	# >= 75: one army of Rome
	has_military_access = {
		var:tfe_unity >= 75
	}
	join_defensive_wars_auto_call = { var:tfe_unity >= 75 }
	# >= 50: the other Augustus can be asked for help
	join_defensive_wars_can_call = { var:tfe_unity >= 50 }

	# < 50: the Augusti may go to war with each other, but only over the succession of the Empire
	can_declare_war = {
		trigger_if = {
			limit = {
				scope:attacker ?= { is_member_of_international_organization = scope:recipient }
				scope:defender ?= { is_member_of_international_organization = scope:recipient }
			}
			scope:recipient.var:tfe_unity < 50
		}
		trigger_else = { always = yes }
	}

	ai_desire_to_attack_other_members = {
		if = {
			limit = { scope:recipient.var:tfe_unity >= 35 }
			subtract = { desc = "TFE_UNITY_HOLDS" value = 10000 }
		}
	}

	modifier = {
		monthly_legitimacy = 0.05
	}
	modifier = {
		potential_trigger = { scope:recipient.var:tfe_unity >= 75 }
		monthly_prestige = 0.1
	}

	auto_disband_trigger = {
		OR = {
			var:tfe_unity <= 0
			total_members < 2
		}
	}
}

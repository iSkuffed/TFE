# TFE: when the action bar buttons show and work (scripted_effects/tfe_cmf.txt does what they do). The name must equal the element.
tfe_illyricum_claims = {
	scope = country
	is_shown = {
		tag = WRE
		country_exists = c:EAR
	}
	is_valid = {
		NOT = { has_variable = tfe_illyricum_claim }
	}
}

tfe_align_visigoths = {
	scope = country
	is_shown = {
		tag = WRE
		country_exists = c:VIS
	}
	is_valid = {
		NOT = { has_variable = tfe_visigoths_aligned }
	}
}

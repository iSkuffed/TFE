# TFE: a migrating host (generic_actions/tfe_migratory.txt) has come for land. Without this it could only
# declare a no-CB war: a heavy stability hit for the whole point of migrating. Start Migration grants it against
# the Roman empires (the game never offers it to a landless host on its own).
cb_tfe_migration = {
	create_visible = {
		has_variable = tfe_migrating
		NOT = { any_owned_location = { always = yes } }
	}
	create_enabled = {
		scope:target = { any_owned_location = { always = yes } }
	}
	speed = 100

	war_goal_type = superiority_tfe_migration
}

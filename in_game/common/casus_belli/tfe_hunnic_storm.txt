# TFE: the wars of the Hunnic Storm (situations/tfe_hunnic_storm.txt). Their treaties: peace_treaties/tfe_hunnic_storm.txt.
# The Huns bring a neighbour of the Yoke under it
cb_tfe_submit_to_the_yoke = {
	create_visible = {
		tag = HNS
	}
	create_enabled = {
		scope:target = {
			NOT = { tfe_is_under_the_yoke = yes }
			NOT = { tag = WRE }
			NOT = { tag = EAR }
			any_neighbor_country = { tfe_is_under_the_yoke = yes }
		}
	}
	speed = 100
	war_goal_type = tfe_submit_to_the_yoke
}

# The Huns ride out for plunder: the loser joins the Yoke and pays its tribute. Only the Grand Raid button
# (generic_actions/tfe_hunnic_raid.txt) grants it, against a neighbour, for prestige.
cb_tfe_grand_raid = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	speed = 100
	war_goal_type = tfe_grand_raid
}

# A member of the Yoke rises against the Huns: only from the Break the Yoke action or the Reckoning's rising event
cb_tfe_break_the_yoke = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	war_goal_type = superiority
	additional_war_enthusiasm_attacker = 0.75
}

# While the Scourge rides, anyone outside the Yoke may take the field against the Huns
cb_tfe_stand_against_the_scourge = {
	create_visible = {
		NOT = { tfe_is_under_the_yoke = yes }
		situation:tfe_hunnic_storm = {
			situation_is_active = yes
			has_variable = tfe_storm_phase
			var:tfe_storm_phase = 2
		}
	}
	create_enabled = {
		scope:target = { tag = HNS }
	}
	speed = 100
	war_goal_type = superiority
}

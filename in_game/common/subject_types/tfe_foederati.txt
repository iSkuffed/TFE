# TFE: foederati - a people sworn to fight for an Augustus. They pay no tribute; Rome pays them (the annona)
# and they march in all of Rome's wars. The foedus binds two men, not two peoples: when either ruler dies it
# lapses (on_action/tfe_foederati.txt) and the foederati choose to swear anew or take what was promised.
# Works for army-based hosts too, landed or landless (tested in game 2026-09-27).
tfe_foederati = {
	color = subject_tributary
	level = 0
	annullment_favours_required = 5

	# only an Augustus can strike a foedus, and not with the other half of the Empire
	visible = {
		any_international_organizations_member_of = {
			international_organization_type = international_organization_type:tfe_roman_empire
		}
		scope:target = {
			subject_type_is_not_locked = yes
			NOT = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tfe_roman_empire
				}
			}
		}
	}
	creation_visible = {
		any_international_organizations_member_of = {
			international_organization_type = international_organization_type:tfe_roman_empire
		}
	}

	# the foederati march in all of Rome's wars; Rome does not march in theirs
	join_offensive_wars_always = { NOT = { scope:actor ?= { is_subject_of = scope:recipient } } }
	join_defensive_wars_always = { NOT = { scope:actor ?= { is_subject_of = scope:recipient } } }

	# the annona: Rome feeds and pays its foederati
	on_monthly = {
		overlord ?= { add_gold = -2 }
		add_gold = 2
	}

	has_overlords_ruler = no
	will_join_independence_wars = yes
	can_be_annexed = no
	strength_vs_overlord = -0.25
	subject_can_cancel = no		# an oath is not cancelled, it is broken (tfe_foederati.1) or fought free of
	overlord_can_cancel = yes	# Rome may stop paying
	allow_declaring_wars = { always = yes }
	can_change_rank = yes
	can_change_heir_selection = yes
	has_limited_diplomacy = no
	diplomatic_capacity_cost_scale = 0.5
	use_overlord_map_color = no
	use_overlord_map_name = no
	great_power_score_transfer = 0.25
	maritime_path_tolerance = 0.25
	institution_spread_to_overlord = monthly_institution_spread_weak
	institution_spread_to_subject = monthly_institution_spread_weak
	minimum_opinion_for_offer = 50

	overlord_modifier = {
		monthly_prestige = 0.01
	}
	subject_modifier = {
	}

	# a paid war band says yes more easily than a tributary
	diplo_chance_accept_subject = {
		base = -30
		current_strength = 0.1
		border_distance = -0.3
		negative_opinion = -5
		positive_opinion = 0.25
		rank_difference = -2
		actor_at_war = 10
		competing_power = -200
		trust_in_actor = 0.2
		reject_subjugation_reasons = -1
		accept_subjugation_reasons = 1
	}
	diplo_chance_accept_overlord = {
		base = 9
		border_distance = -0.3
		negative_opinion = -5
		positive_opinion = 0.25
		competing_power = -200
		trust_in_actor = 0.2
	}
}

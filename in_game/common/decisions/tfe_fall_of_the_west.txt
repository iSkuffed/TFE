# honorius-only: Stilicho's Claims is Honorius's (tools/test_western_rome.py)
# TFE: the Fall of the West's decisions (written by script/decisions.py).
# Migrate: a people of Germania or Dacia (tfe_is_migrator) may give up its homeland, while the Decline of the West
# runs, for a great host that costs nothing while it owns no land (auto_modifiers/tfe_migratory.txt). It cannot
# replenish: landless, it has no manpower, so every warrior lost is gone until it wins land of its own. One decision per
# empire: the host marches on the East or the West at once, and that Augustus may buy it off with land (Hospitalitas,
# generic_actions/tfe_decline_rome.txt). The two differ only in the empire. A settled host may take the road again until
# it reforms into a monarchy (decisions/tfe_barbarian_kingdoms.txt). The Salian Franks never migrate: they Cross the Rhine.
# Stilicho's Claims: Honorius's West takes a casus belli on Eastern Illyricum, at a cost in Unity.
tfe_migrate_east = {
	decision_category = tfe_fall_of_the_west
	image = "gfx/interface/illustrations/event/backgrounds/exterior/soldiers/north_german_soldiers_exterior.dds"
	potential = {
		situation:tfe_decline_of_the_west = { situation_is_active = yes }
		tfe_is_migrator = yes
		NOT = { tag = SLF }	# the Salians raid over the Rhine instead (decisions/tfe_barbarian_kingdoms.txt)
		OR = {
			NOT = { has_variable = tfe_migrating }
			has_variable = tfe_settled	# a settled host may take the road again
		}
		NOT = { has_variable = tfe_host_kingdom }	# until it reforms into a monarchy (decisions/tfe_barbarian_kingdoms.txt)
		country_exists = c:EAR
	}
	allow = {
		custom_tooltip = {
			text = tfe_migration_not_yet_tt
			current_date >= 395.2.18	# a month after the game starts (defines' START_DATE)
		}
		is_subject = no	# no warband afield is fine: the host gathers at the capital
		tfe_frontier_unmanned = yes	# scripted_triggers/tfe_decline_rome.txt: a manned frontier holds them
	}
	# Pace the chaos: one AI people on the road at a time. None moves within a year of the last host setting out
	# (tfe_host_took_the_road), and then only when pushed: the Huns at the door or Rome divided. A settled AI host
	# stays settled.
	ai_will_do = {
		value = 0
		if = {
			limit = { NOT = { has_global_variable = tfe_host_took_the_road } at_war = no NOT = { has_variable = tfe_settled } }
			add = 5
			if = {
				limit = { any_neighbor_country = { tfe_is_under_the_yoke = yes } }
				add = 45
			}
			# the empire across the river first
			if = {
				limit = { any_neighbor_country = { tag = EAR } }
				add = 10
			}
			if = {
				limit = { international_organization:tfe_roman_empire ?= { var:tfe_unity < 50 } }
				add = 25
			}
		}
	}
	option = {
		name = tfe_migrate_east.a
		ai_chance = { value = 100 }
		effect = {
			custom_tooltip = tfe_start_migration_tt
			custom_tooltip = tfe_migrate_EAR_tt
			hidden_effect = {
				tfe_start_migration_effect = yes	# scripted_effects/tfe_migratory.txt
				declare_war_with_cb = { target = c:EAR type = casus_belli:cb_tfe_migration }
			}
		}
	}
}

tfe_illyricum_claims = {
	decision_category = tfe_fall_of_the_west
	image = "gfx/interface/illustrations/event/backgrounds/exterior/byz_soldiers_exterior.dds"
	potential = {
		exists = international_organization:tfe_roman_empire
		tag = WRE
		country_exists = c:EAR
	}
	allow = {
		custom_tooltip = {
			text = tfe_stilicho_serves_us_tt
			tfe_stilicho_serves_us = yes
		}
		# ten years between claims; no Unity floor: driving Unity down is the point
		custom_tooltip = {
			text = tfe_illyricum_claim_pressed_tt
			NOT = { has_variable = tfe_illyricum_claim }
		}
	}
	ai_will_do = {
		# the player's choice: the AI West never picks a quarrel with the East this way
		value = -1
	}
	option = {
		name = tfe_illyricum_claims.a
		ai_chance = { value = 100 }
		effect = {
			set_variable = { name = tfe_illyricum_claim years = 10 }
			add_casus_belli = { target = c:EAR type = casus_belli:cb_tfe_illyrian_claim years = 10 }
			custom_tooltip = tfe_unity_down_10_tt
			hidden_effect = {
				international_organization:tfe_roman_empire ?= {
					change_variable = { name = tfe_unity add = -10 }
				}
			}
		}
	}
}

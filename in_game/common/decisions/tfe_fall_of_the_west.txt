# honorius-only: Stilicho's Claims is Honorius's (tools/test_western_rome.py)
# TFE: the Fall of the West's Native Decisions (written by script/decisions.py). Migrate into Rome
# (generic_actions/tfe_fall_of_the_west.txt, shown in the decisions tab): a people of Germania or Dacia (tfe_is_migrator) may
# give up its homeland, while the Decline of the West runs, for a great host that costs nothing while it owns no land
# (auto_modifiers/tfe_migratory.txt). Landless, it has no manpower: only the Roman towns it takes send it men
# (on_action/tfe_defectors.txt). The host marches at once on a province of the Roman state it chooses; that ruler may buy
# it off with land (Hospitalitas, generic_actions/tfe_decline_rome.txt). Won battles win the war, the peace hands over the
# province, and a host that loses breaks up (on_action/tfe_migratory.txt, events/tfe_barbarian_kingdoms.txt). A settled host may take the road again until it reforms into a
# monarchy (decisions/tfe_barbarian_kingdoms.txt). The Salian Franks never migrate: they Cross the Rhine.
# Stilicho's Claims: Honorius's West takes a casus belli on Eastern Illyricum, at a cost in Unity.
tfe_illyricum_claims = {
	decision_category = tfe_fall_of_the_west
	image = "gfx/interface/illustrations/event/backgrounds/exterior/byz_soldiers_exterior.dds"
	potential = {
		exists = international_organization:tfe_roman_empire
		tag = WRE
		country_exists = c:EAR
	}
	allow = {
		custom_tooltip = {
			text = tfe_stilicho_serves_us_tt
			tfe_stilicho_serves_us = yes
		}
		# ten years between claims; no Unity floor: driving Unity down is the point
		custom_tooltip = {
			text = tfe_illyricum_claim_pressed_tt
			NOT = { has_variable = tfe_illyricum_claim }
		}
	}
	ai_will_do = {
		# the player's choice: the AI West never picks a quarrel with the East this way
		value = -1
	}
	option = {
		name = tfe_illyricum_claims.a
		ai_chance = { value = 100 }
		effect = {
			set_variable = { name = tfe_illyricum_claim years = 10 }
			add_casus_belli = { target = c:EAR type = casus_belli:cb_tfe_illyrian_claim years = 10 }
			custom_tooltip = tfe_unity_down_10_tt
			hidden_effect = {
				international_organization:tfe_roman_empire ?= {
					change_variable = { name = tfe_unity add = -10 }
				}
			}
		}
	}
}

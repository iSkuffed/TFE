# TFE: Gildo's revolt grows over the first years instead of arriving on one date (events/tfe_gildo.txt).
# The West's tfe_gildo_progress starts at 0; a hidden monthly event feeds it and fires the revolt at 100.
on_game_start = {
	on_actions = { tfe_on_start_gildo }
}

tfe_on_start_gildo = {
	effect = {
		c:WRE ?= {
			set_variable = { name = tfe_gildo_progress value = 0 }
			trigger_event_silently = { id = tfe_gildo.1 months = 1 }
		}
	}
}

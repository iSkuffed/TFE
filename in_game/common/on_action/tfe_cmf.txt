# TFE's buttons on the Community Mod Framework's action bar (scripted_effects/tfe_cmf.txt)
cmf_on_mod_registration = {
	on_actions = { tfe_on_register_cmf }
}

tfe_on_register_cmf = {
	effect = {
		tfe_register_cmf = yes
	}
}

# A click on one of our buttons: var:cmf_callback names it
cmf_on_callback = {
	on_actions = { tfe_on_cmf_callback }
}

tfe_on_cmf_callback = {
	effect = {
		tfe_handle_cmf_callback = yes
	}
}

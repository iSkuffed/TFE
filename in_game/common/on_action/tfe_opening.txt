# TFE: the 395 opening (events/tfe_opening.txt). Each date is when it happened; each event checks its people are still alive.
on_game_start = {
	on_actions = { tfe_on_start_opening }
}

tfe_on_start_opening = {
	effect = {
		c:WRE ?= {
			trigger_event_non_silently = tfe_opening.1
			trigger_event_non_silently = { id = tfe_opening.5 days = 4394 }		# Constantine III, early 407
		}
		c:EAR ?= {
			add_country_modifier = { modifier = tfe_rufinus_prefecture years = 5 }	# until the Campus Martius
			trigger_event_non_silently = { id = tfe_opening.2 days = 313 }		# Rufinus, 27 Nov 395
		}
	}
}

# Honorius comes of age and takes the throne; the next day Eucherius is named his heir (events/tfe_opening.txt).
# A day's delay: the game picks a new ruler's heir after crowning him, over any heir set at once.
on_new_ruler = {
	events = {
		delay = { days = 1 }
		tfe_opening.6
	}
}

# TFE: the first land a migrating host (decisions/tfe_fall_of_the_west.txt) wins ends the migration. It becomes a
# landed country again, that land is its capital, and the great host disbands back to the warband it started with
# there; its free upkeep (auto_modifiers/tfe_migratory.txt) ends until it takes the road again. Its people settle
# the land it won.
on_location_changed_owner = {
	on_actions = { tfe_on_host_settles }
}

tfe_on_host_settles = {
	trigger = {
		exists = scope:winner
		scope:winner = {
			has_variable = tfe_migrating
			NOT = { has_variable = tfe_settled }
		}
	}
	effect = {
		save_scope_as = tfe_homeland
		scope:winner = {
			set_variable = tfe_settled
			tfe_list_the_migrators = yes	# the Decline of the West's panel moves it to "settled" now, not next month
			# user: a settled host could not disband its troops. An army-based country lives by its armies, so the game
			# keeps them; vanilla's settling pirates and Timurids become landed countries (change_country_type).
			# User: in a peace deal its people settled anywhere but by the capital. The first land is both.
			change_country_type = location
			set_capital = scope:tfe_homeland
			# the host breaks up and its old warband (setup/395/27_armies.txt) re-forms at home, raised from it.
			# Whole armies go: destroying sub-units one by one crashed the game a tick later.
			every_army = { destroy_unit = yes }
		}
		while = {
			count = 4
			create_sub_unit_with_owner = { type = a_footmen owner = scope:winner origin = scope:tfe_homeland }
		}
		while = {
			count = 2
			create_sub_unit_with_owner = { type = a_tribal_cavalry owner = scope:winner origin = scope:tfe_homeland }
		}
		# the people who followed the host (tfe_host_people, counted by tfe_start_migration_effect) settle a day later,
		# once the whole peace or Hospitalitas has handed over its land, spread over all of it (events/tfe_barbarian_kingdoms.txt)
		scope:winner = { trigger_event_silently = { id = tfe_barbarian_kingdoms.1 days = 1 } }
	}
}

# Rome lost its field armies in a few great defeats: each battle a host on the road wins against a Roman state,
# with 1000 or more of the enemy dead, adds 25 war score in its Migrate into Rome war (vanilla
# scores a battle by the share of the loser's army, so the Empire shrugged off defeats its tiny allies did not).
# root = the winner, scope:target = the losing unit.
on_battle_won = {
	on_actions = { tfe_on_host_wins_battle }
}

on_great_battle_won = {
	on_actions = { tfe_on_host_wins_battle }
}

# a great battle fires on_great_battle_won instead of on_battle_won, never both
tfe_on_host_wins_battle = {
	trigger = {
		has_variable = tfe_migrating
		NOT = { has_variable = tfe_settled }
		scope:killed_land_units >= 1000
		scope:target.owner ?= { tfe_is_roman_state = yes }
	}
	effect = {
		"war_with_country(scope:target.owner)" = {
			# a link with arguments is a quoted key, as vanilla writes it; unquoted the game drops the block
			if = {
				limit = { casus_belli ?= casus_belli:cb_tfe_migration }
				add_bonus_warscore = { country = root amount = 25 }
			}
		}
	}
}

# TFE: in 395 every people from Rome to the steppe raises some tribute. Without Taxation a country has no taxes
# and no stability investment, so every country starts with it and the advances it requires.
on_game_start = {
	on_actions = { tfe_on_start_taxation }
}

tfe_on_start_taxation = {
	effect = {
		every_country = {
			if = {
				limit = { NOT = { has_advance = written_alphabet } }
				research_advance = advance_type:written_alphabet
			}
			if = {
				limit = { NOT = { has_advance = codified_laws } }
				research_advance = advance_type:codified_laws
			}
			if = {
				limit = { NOT = { has_advance = taxation_advance } }
				research_advance = advance_type:taxation_advance
			}
		}
		c:HNS ?= {
			if = {
				limit = { NOT = { has_advance = agriculture_advance } }
				research_advance = advance_type:agriculture_advance
			}
			if = {
				limit = { NOT = { has_advance = windmills_advance } }
				research_advance = advance_type:windmills_advance
			}
			if = {
				limit = { NOT = { has_advance = ranching } }
				research_advance = advance_type:ranching
			}
			if = {
				limit = { NOT = { has_advance = horse_riding_advance } }
				research_advance = advance_type:horse_riding_advance
			}
			if = {
				limit = { NOT = { has_advance = horse_lords } }
				research_advance = advance_type:horse_lords
			}
		}
	}
}

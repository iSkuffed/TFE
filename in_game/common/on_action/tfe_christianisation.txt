tfe_on_random_missionary = {
	trigger = {
		NOT = { has_variable = tfe_mission_to }
		num_locations >= 3
		OR = {
			AND = {
				religion = religion:orthodox
				tfe_nicene_mission_in_reach = yes
			}
			AND = {
				religion = religion:arianism
				tfe_arian_mission_in_reach = yes
			}
		}
	}
	effect = {
		if = {
			limit = { has_country_modifier = tfe_temples_closed }
			random = {
				chance = 10
				if = {
					limit = { religion = religion:orthodox }
					every_owned_location = {
						if = {
							limit = { tfe_nicene_mission_field = yes }
							add_to_temporary_list = tfe_reach_fields
						}
						every_neighbor_location = {
							if = {
								limit = { tfe_nicene_mission_field = yes }
								add_to_temporary_list = tfe_reach_fields
							}
						}
					}
					random_in_list = {
						weight = {
							base = 1
							modifier = {
								add = population
							}
						}
						list = tfe_reach_fields
						save_scope_as = tfe_mission_to
					}
					if = {
						limit = { exists = scope:tfe_mission_to }
						capital = { save_scope_as = tfe_mission_from }
						create_character = {
							age = 35
							religion = root.religion
							estate = estate_type:clergy_estate
							culture = root.culture
							save_scope_as = tfe_missionary
						}
						if = {
							limit = { exists = scope:tfe_missionary }
							tfe_send_missionary_effect = yes
						}
					}
				}
				if = {
					limit = { religion = religion:arianism }
					every_owned_location = {
						if = {
							limit = { tfe_arian_mission_field = yes }
							add_to_temporary_list = tfe_reach_fields
						}
						every_neighbor_location = {
							if = {
								limit = { tfe_arian_mission_field = yes }
								add_to_temporary_list = tfe_reach_fields
							}
						}
					}
					random_in_list = {
						weight = {
							base = 1
							modifier = {
								add = population
							}
						}
						list = tfe_reach_fields
						save_scope_as = tfe_mission_to
					}
					if = {
						limit = { exists = scope:tfe_mission_to }
						capital = { save_scope_as = tfe_mission_from }
						create_character = {
							age = 35
							religion = root.religion
							estate = estate_type:clergy_estate
							culture = root.culture
							save_scope_as = tfe_missionary
						}

tfe_on_missions_end = {
	effect = {
		every_character = {
			limit = { has_variable = tfe_mission_at }
			# a character's timed variable ran a month for every year in game (saves at 400 and 411): count by hand
			change_variable = { name = tfe_preaching add = -1 }
			if = {
				limit = { var:tfe_preaching <= 0 }
				save_scope_as = tfe_missionary
				var:tfe_mission_at = { save_scope_as = tfe_mission_from }
				root = {
					tfe_end_mission_effect = yes
					if = {
						limit = {
							NOT = { has_variable = tfe_mission_to }
							scope:tfe_missionary = {
								religion = religion:orthodox
								var:tfe_missions < 3
							}
						}
						random = {
							chance = 50
							scope:tfe_mission_from = {
								every_neighbor_location = {
									if = {
										limit = { tfe_nicene_mission_field = yes NOT = { this = scope:tfe_mission_from } }
										add_to_temporary_list = tfe_near_fields
									}
									every_neighbor_location = {
										if = {
											limit = { tfe_nicene_mission_field = yes NOT = { this = scope:tfe_mission_from } }
											add_to_temporary_list = tfe_near_fields
										}
										every_neighbor_location = {
											if = {
												limit = { tfe_nicene_mission_field = yes NOT = { this = scope:tfe_mission_from } }
												add_to_temporary_list = tfe_near_fields
											}
										}
									}
								}
							}
							ordered_in_list = {
								order_by = population
								list = tfe_near_fields
								save_scope_as = tfe_mission_to
							}
							if = {
								limit = { exists = scope:tfe_mission_to }
								tfe_send_missionary_effect = yes
							}
						}
					}
					if = {
						limit = {
							NOT = { has_variable = tfe_mission_to }
							scope:tfe_missionary = {
								religion = religion:arianism
								var:tfe_missions < 3
							}
						}
						random = {
							chance = 50
							scope:tfe_mission_from = {
								every_neighbor_location = {
									if = {
										limit = { tfe_arian_mission_field = yes NOT = { this = scope:tfe_mission_from } }
										add_to_temporary_list = tfe_near_fields
									}
									every_neighbor_location = {
										if = {
											limit = { tfe_arian_mission_field = yes NOT = { this = scope:tfe_mission_from } }
											add_to_temporary_list = tfe_near_fields
										}
										every_neighbor_location = {
											if = {
												limit = { tfe_arian_mission_field = yes NOT = { this = scope:tfe_mission_from } }
												add_to_temporary_list = tfe_near_fields
											}
										}
									}
								}
							}
							ordered_in_list = {
								order_by = population
								list = tfe_near_fields
								save_scope_as = tfe_mission_to
							}
							if = {
								limit = { exists = scope:tfe_mission_to }
								tfe_send_missionary_effect = yes
							}
						}
					}
					# a kill in the character's own scope fails PostValidate (probe): kill from the country
					if = {
						limit = { scope:tfe_missionary = { NOT = { has_variable = tfe_on_road } } }
						kill_character_silently = scope:tfe_missionary
					}
				}
			}
		}
	}
}

	value = 0
	international_organization:tfe_hunnic_yoke ?= {
		every_international_organization_member = {
			add = 1
		}
	}
	c:HNS ?= {
		every_subject = {
			limit = {
				is_subject_type = tributary
				NOT = { is_member_of_international_organization = international_organization:tfe_hunnic_yoke }
			}
			add = 1
		}
	}
}

# TFE: the West's burdens in 395. Two privileges the estates hold over the crown, granted at start (10_countries).
# Neither can be revoked until its reform (government_reforms/tfe_late_roman_west.txt) is done; once revoked,
# neither can be granted again.

# The clarissimi: senators excused the sordid munera, their land-tax arrears remitted, their estates beyond the census
tfe_senatorial_immunities = {
	estate = nobles_estate

	potential = {
		has_or_had_tag = WRE
		NOT = { has_variable = tfe_senatorial_immunities_revoked }
	}
	can_revoke = {
		custom_tooltip = {
			text = tfe_peraequatio_done_tt
			has_variable = tfe_peraequatio_done
		}
	}
	on_deactivate = {
		set_variable = { name = tfe_senatorial_immunities_revoked }
	}

	country_modifier = {
		global_nobles_estate_power = 0.5
		nobles_estate_target_satisfaction = medium_privilege_target_satisfaction
		nobles_estate_max_tax = -0.25
	}
}

# Patrocinium: coloni shelter under a great landlord, who hides them from the recruiting officer and pays gold instead
tfe_patrocinium = {
	estate = peasants_estate

	potential = {
		has_or_had_tag = WRE
		NOT = { has_variable = tfe_patrocinium_revoked }
	}
	can_revoke = {
		custom_tooltip = {
			text = tfe_dilectus_done_tt
			has_variable = tfe_dilectus_done
		}
	}
	on_deactivate = {
		set_variable = { name = tfe_patrocinium_revoked }
	}

	country_modifier = {
		global_peasants_estate_power = 0.25
		global_nobles_estate_power = 0.25	# the patron is a great landlord
		peasants_estate_target_satisfaction = medium_privilege_target_satisfaction
		peasants_estate_levy_size = -0.5
		global_manpower_modifier = -0.25
	}
}

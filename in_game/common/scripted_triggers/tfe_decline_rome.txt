# TFE: Rome's answers to the migrations (generic_actions/tfe_decline_rome.txt)
# A location (scope: location) that holds the frontier works placed at start (building_types/tfe_frontier.txt)
tfe_has_frontier_works = {
	any_buildings_in_location = {
		OR = {
			building_type = building_type:tfe_limes
			building_type = building_type:tfe_hadrians_wall
		}
	}
}

# A country (scope: country) with no land beside a stretch of frontier that an Augustus has manned (Man the Limes).
# True when the way is open, so the Migrate buttons' requirement reads right with either mark: a custom_tooltip shows
# its text as written, and "= no" on a barred-trigger showed "A Roman garrison mans the frontier" ticked green.
tfe_frontier_unmanned = {
	custom_tooltip = {
		text = tfe_frontier_unmanned_tt
		NOT = {
			any_owned_location = {
				any_neighbor_location = { has_location_modifier = tfe_limes_manned }
			}
		}
	}
}

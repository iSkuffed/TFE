# TFE: Britain's army crowns a common soldier for his name (early 407, tfe_opening.5; or when Stilicho rises).
# A revolt, not a gift of land, as Gildo's was: the West is at war with the revolter at once, and the war screen's
# Annex Revolter takes Britain back. tfe_opening.9 makes the revolter Constantine's empire in the same moment.
tfe_constantine_rises = {
	create_rebel = {
		category = nationalist
		name = tfe_constantine_rebels
		religion = religion:orthodox
		culture = culture:roman_culture
		save_scope_as = tfe_constantine_rebels
	}
	every_owned_location = {
		limit = { region = region:great_britain_region }
		every_pop = { change_pop_allegiance = scope:tfe_constantine_rebels }
	}
	# the wars we are in already (a host, Gildo, Stilicho's revolt) are not his: mark them, then find the new one
	every_country = {
		limit = { is_at_war_with = root }
		set_variable = tfe_old_enemy
	}
	scope:tfe_constantine_rebels ?= { start_revolt = yes }
	random_country = {
		limit = {
			is_at_war_with = root
			NOT = { has_variable = tfe_old_enemy }
			# no backer: the revolter holds nothing beyond Britain, or nothing at all
			NOT = { any_owned_location = { NOT = { region = region:great_britain_region } } }
		}
		set_variable = tfe_constantine_revolter
	}
	every_country = {
		limit = { has_variable = tfe_old_enemy }
		remove_variable = tfe_old_enemy
	}
	trigger_event_silently = tfe_opening.9
}

# the diocese of Africa goes its own way under a Roman count, when Stilicho rises and Gildo's kingdom is gone
tfe_africa_breaks_away = {
	random_owned_location = {
		limit = { region = region:maghreb_region }
		create_country_from_location = {
			define_unique_country_tag = AFRIC
			change_country_name = TFE_AFRICA
			change_country_adjective = TFE_AFRICA_ADJ
			change_country_color = map_kado
			change_country_flag = TFE_AFRICA
			set_country_rank = country_rank:rank_kingdom	# a count of Africa, not an Augustus
			change_government_type = government_type:monarchy
			add_gold = 200
			create_character = {
				adm = 50
				age = 50
				religion = religion:orthodox
				estate = estate_type:nobles_estate
				culture = culture:roman_culture
				mil = 45
				save_scope_as = tfe_usurper_ruler
			}
			set_new_ruler = scope:tfe_usurper_ruler
			set_variable = { name = tfe_usurper_against value = root years = 10 }
			save_scope_as = tfe_usurper
		}
	}
	every_owned_location = {
		limit = { region = region:maghreb_region }
		change_location_owner = scope:tfe_usurper
	}
}

# TFE: action bar buttons through the Community Mod Framework (the workshop mod, a dependency in .metadata). Scope: country.
tfe_register_cmf = {
	cmf_register_mod = { mod_id = tfe }
	cmf_add_action_bar_element = { element = tfe_illyricum_claims }
	cmf_register_scripted_gui = { element = tfe_illyricum_claims }
	cmf_add_action_bar_element = { element = tfe_align_visigoths }
	cmf_register_scripted_gui = { element = tfe_align_visigoths }
}

tfe_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		flag:tfe_illyricum_claims = {
			# Stilicho's claim to Eastern Illyricum: a casus belli against the East for 10 years, at the cost of Unity
			set_variable = { name = tfe_illyricum_claim years = 10 }
			international_organization:tfe_roman_empire ?= { change_variable = { name = tfe_unity add = -10 } }
			add_casus_belli = { target = c:EAR type = casus_belli:cb_tfe_illyrian_claim years = 10 }
		}
		flag:tfe_align_visigoths = {
			# Alaric's Goths bought out of the Eastern foedus and allied to the West: they march in the West's wars
			set_variable = tfe_visigoths_aligned
			international_organization:tfe_roman_empire ?= { change_variable = { name = tfe_unity add = -10 } }
			c:VIS ?= {
				if = {
					limit = { is_subject_type = tfe_foederati }
					overlord = { cancel_subject = c:VIS }
				}
			}
			create_relation = { first = root second = c:VIS type = relation_type:alliance }
		}
	}
}

# TFE: a host takes to the road (the Migrate decisions, decisions/tfe_fall_of_the_west.txt; the Huns push
# the Germanic peoples west in events/tfe_hunnic_storm.txt). Scope: the migrating country. It becomes an army-based
# country, musters a great host, abandons its homeland and gets a casus belli on the Roman empires: a landless host
# has no neighbours for the game to offer one against. A fifth of its people go with it, to settle the land it wins
# (on_action/tfe_migratory.txt). Its passage over the land it leaves is Barbaricum's (international_organizations/).
tfe_start_migration_effect = {
	save_scope_as = tfe_host
	set_variable = tfe_migrating
	# pace the chaos: the AI's next host waits a year (the decisions' ai_will_do)
	set_global_variable = { name = tfe_host_took_the_road value = yes years = 1 }
	# only the Vandals start army-based; a landed people would be gone with its last location (vanilla CHB, OIR)
	if = {
		limit = { NOT = { country_type = army } }
		change_country_type = army
	}
	tfe_muster_the_host_effect = yes	# scripted_effects/tfe_barbarian_kingdoms.txt: 24 footmen and 8 tribal cavalry
	# a fifth of our people take the road; the rest stay under the new lords. Counted by pop_size, whatever its unit.
	set_variable = { name = tfe_host_people value = 0 }	# change_variable fails on an unset variable
	every_owned_location = {
		every_pop = {
			limit = { culture = scope:tfe_host.culture }
			save_temporary_scope_as = tfe_leaver
			scope:tfe_host = {
				change_variable = { name = tfe_host_people add = { value = scope:tfe_leaver.pop_size multiply = 0.2 } }
			}
			add_pop_size = { value = pop_size multiply = -0.2 }
		}
	}
	# the homeland is given up for good: the host keeps no core on it, nor on any land it won before
	every_core_location = { remove_core = scope:tfe_host }
	# those who stayed make a country of the homeland, a tribe of the majority culture at the old capital, in the
	# host's faith, holding every location the host leaves; no neighbour gets it. A host that owns no capital leaves
	# nothing to hold: the land is abandoned. The remnant takes its name from the game's default for a new country.
	if = {
		limit = { capital ?= { owner = scope:tfe_host } }
		capital = {
			save_scope_as = tfe_old_capital
			create_country_from_location = {
				change_government_type = government_type:tribe	# only counts after the effect block ends
				change_culture = scope:tfe_old_capital.dominant_culture
				change_religion = scope:tfe_host.religion
				set_variable = { name = tfe_remnant_of value = scope:tfe_host }
				save_scope_as = tfe_remnant
			}
		}
		every_owned_location = {
			change_location_owner = scope:tfe_remnant
			add_core = scope:tfe_remnant
		}
	}
	else = {
		every_owned_location = { scope:tfe_host = { abandon_location = prev } }
	}
	if = {
		limit = { country_exists = c:EAR }
		add_casus_belli = { target = c:EAR type = casus_belli:cb_tfe_migration }
	}
	# and on every western Rome: Stilicho's West too, once he rises
	every_country = {
		limit = { tfe_is_western_rome = yes }
		scope:tfe_host = {
			add_casus_belli = { target = prev type = casus_belli:cb_tfe_migration }
		}
	}
	# a settled host taking the road again is on the road once more: free, fed by Roman towns, settling anew
	if = {
		limit = { has_variable = tfe_settled }
		remove_variable = tfe_settled
	}
	tfe_list_the_migrators = yes
}

# The Decline of the West's panel (gui/panels/situation/tfe_decline_of_the_west.gui) lists the migrators by state:
# at home, on the road (tfe_migrating), settled (tfe_settled). Any scope; the situation refreshes it monthly.
tfe_list_the_migrators = {
	situation:tfe_decline_of_the_west = {
		if = {
			limit = { has_variable_list = tfe_migrators_at_home }
			clear_variable_list = tfe_migrators_at_home
		}
		if = {
			limit = { has_variable_list = tfe_migrators_on_the_road }
			clear_variable_list = tfe_migrators_on_the_road
		}
		if = {
			limit = { has_variable_list = tfe_migrators_settled }
			clear_variable_list = tfe_migrators_settled
		}
	}
	every_country = {
		limit = { tfe_is_migrator = yes }
		save_scope_as = tfe_migrator
		situation:tfe_decline_of_the_west = {
			if = {
				limit = { scope:tfe_migrator = { has_variable = tfe_settled } }
				add_to_variable_list = { name = tfe_migrators_settled target = scope:tfe_migrator }
			}
			else_if = {
				limit = { scope:tfe_migrator = { has_variable = tfe_migrating } }
				add_to_variable_list = { name = tfe_migrators_on_the_road target = scope:tfe_migrator }
			}
			else = {
				add_to_variable_list = { name = tfe_migrators_at_home target = scope:tfe_migrator }
			}
		}
	}
}

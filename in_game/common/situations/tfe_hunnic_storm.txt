# TFE: the Hunnic Storm (RoadMap #10). The Huns build their tribute empire, the Hunnic Yoke
# (international_organizations/tfe_hunnic_yoke.txt), may crown a Scourge of God, and face the Reckoning when he dies.
# Its phase is var:tfe_storm_phase (1 Stirring, 2 Scourge, 3 Reckoning); each opens on a cause the Huns earn, with a
# dated fallback only as a nudge. Started on day one by on_action/tfe_hunnic_storm.txt; events/tfe_hunnic_storm.txt.
tfe_hunnic_storm = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	can_start = {
		situation_has_ended = no
		country_exists = c:HNS
		current_date < 480.1.1
	}
	can_end = {
		tfe_hunnic_storm_is_over = yes
	}
	visible = {
		OR = {
			tag = HNS
			tag = WRE
			tag = EAR
			tfe_is_under_the_yoke = yes
			any_neighbor_country = { tfe_is_under_the_yoke = yes }
		}
	}

	on_start = {
		set_variable = { name = tfe_storm_phase value = 1 }
	}

	on_monthly = {
		# Phase 2, the Scourge: a Yoke of 8 and a Roman empire in the Yoke; from 434, a Yoke of 6 suffices
		if = {
			limit = {
				var:tfe_storm_phase = 1
				OR = {
					AND = {
						tfe_yoke_size >= 8
						international_organization:tfe_hunnic_yoke ?= {
							any_international_organization_member = { OR = { tag = WRE tag = EAR } }
						}
					}
					AND = {
						current_date >= 434.1.1
						tfe_yoke_size >= 6
					}
				}
			}
			set_variable = { name = tfe_storm_phase value = 2 }
			if = {
				limit = { international_organization:tfe_hunnic_yoke ?= { any_international_organization_member = { OR = { tag = WRE tag = EAR } } } }
				set_variable = { name = tfe_storm_by_subsidy value = yes }
			}
			# the fallback Reckoning: any Hunnic ruler's death once this runs out (on_action/tfe_hunnic_storm.txt)
			set_variable = { name = tfe_scourge_clock value = yes years = 10 }
			every_country = {
				limit = { can_see_situation = situation:tfe_hunnic_storm }
				trigger_event_non_silently = tfe_hunnic_storm.1
			}
			# the peoples in the Huns' path must move or submit: hosts that can take to the road are asked
			every_country = {
				limit = {
					OR = {
						tag = GEP
						tag = SCR
						tag = RUG
						tag = HAS
						tag = IAZ
						tag = QAD
					}
					country_type = army
					NOT = { has_variable = tfe_migrating }
					NOT = { tfe_is_under_the_yoke = yes }
					any_neighbor_country = { tfe_is_under_the_yoke = yes }
				}
				trigger_event_non_silently = tfe_hunnic_storm.2
			}
		}

		# Phase 3, the Reckoning: the Scourge of God is dead, or (fallback) a Hunnic ruler died after 10 years of phase 2
		else_if = {
			limit = {
				var:tfe_storm_phase = 2
				OR = {
					has_variable = tfe_reckoning_due
					AND = {
						has_variable = tfe_scourge
						var:tfe_scourge = { is_alive = no }
					}
				}
			}
			set_variable = { name = tfe_storm_phase value = 3 }
			set_variable = { name = tfe_reckoning_clock value = yes years = 10 }
			# the Yoke is broken once it has lost half its subject peoples: count them now
			# tfe_reckoning_peoples is also shown on the situation panel: "half of the N subject peoples"
			set_variable = { name = tfe_reckoning_peoples value = 0 }
			every_country = {
				limit = {
					tfe_is_under_the_yoke = yes
					NOT = { tag = HNS }
				}
				root = { change_variable = { name = tfe_reckoning_peoples add = 1 } }
				trigger_event_non_silently = tfe_hunnic_storm.3
			}
			set_variable = { name = tfe_reckoning_threshold value = var:tfe_reckoning_peoples }
			change_variable = { name = tfe_reckoning_threshold multiply = 0.5 }
			change_variable = { name = tfe_reckoning_threshold add = 1 }	# the Huns themselves
		}

		# The Reckoning resolves: the Yoke is broken (Nedao), or the Huns endure its 10 years
		else_if = {
			limit = {
				var:tfe_storm_phase = 3
				OR = {
					NOT = { exists = international_organization:tfe_hunnic_yoke }
					tfe_yoke_size <= situation:tfe_hunnic_storm.var:tfe_reckoning_threshold
				}
			}
			set_variable = { name = tfe_storm_resolved value = yes }
			if = {
				limit = { exists = international_organization:tfe_hunnic_yoke }
				c:HNS ?= { destroy_international_organization = { target = international_organization:tfe_hunnic_yoke } }
			}
			every_country = {
				limit = { can_see_situation = situation:tfe_hunnic_storm }
				trigger_event_non_silently = tfe_hunnic_storm.4
			}
		}
		else_if = {
			limit = {
				var:tfe_storm_phase = 3
				NOT = { has_variable = tfe_reckoning_clock }
			}
			set_variable = { name = tfe_storm_resolved value = yes }
			c:HNS ?= { add_country_modifier = { modifier = tfe_huns_endure years = 30 } }
			every_country = {
				limit = { can_see_situation = situation:tfe_hunnic_storm }
				trigger_event_non_silently = tfe_hunnic_storm.5
			}
		}

		if = {
			limit = { tfe_hunnic_storm_is_over = yes }
			end_situation = situation:tfe_hunnic_storm
		}
	}

	tooltip = {
		if = {
			limit = { owner ?= c:HNS }
			custom_tooltip = TFE_STORM_HUNS
		}
		else_if = {
			limit = { owner ?= { OR = { tag = WRE tag = EAR } is_member_of_international_organization = international_organization:tfe_hunnic_yoke } }
			custom_tooltip = TFE_STORM_SUBSIDY
		}
		else_if = {
			limit = { owner ?= { tfe_is_under_the_yoke = yes } }
			custom_tooltip = TFE_STORM_YOKE
		}
	}

	map_color = {
		if = {
			limit = { owner ?= c:HNS }
			value = rgb { 90 35 20 }
		}
		else_if = {
			limit = { owner ?= { tfe_is_under_the_yoke = yes } }
			value = rgb { 200 140 90 }
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}
	# Romans paying the subsidy are striped in the Huns' colour
	secondary_map_color = {
		if = {
			limit = { owner ?= { OR = { tag = WRE tag = EAR } is_member_of_international_organization = international_organization:tfe_hunnic_yoke } }
			value = rgb { 90 35 20 }
		}
	}

	legend_key = {
		desc = "TFE_STORM_HUNS"
		color = rgb { 90 35 20 }
		require_color_on_map = yes
	}
	legend_key = {
		desc = "TFE_STORM_YOKE"
		color = rgb { 200 140 90 }
		require_color_on_map = yes
	}
	legend_key = {
		desc = "TFE_STORM_SUBSIDY"
		color = rgb { 90 35 20 }
	}
}

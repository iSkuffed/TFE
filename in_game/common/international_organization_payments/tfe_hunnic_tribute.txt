# TFE: every member of the Hunnic Yoke pays the Huns each month, about 1% of the members' combined trade and tax
# income, each in proportion to its own (vanilla tatar_yoke_contribution, paid to the leader, no slider).
tfe_hunnic_tribute = {
	get_payer_list = {
		every_international_organization_member = {
			limit = {
				NOT = {
					is_leader_of_international_organization = root
				}
			}

			add_to_local_variable_list = {
				name = payers
				target = this
			}
		}
	}
	get_payee_list = {
		leader_country ?= {
			add_to_local_variable_list = {
				name = payees
				target = this
			}
		}
	}
	price = tfe_hunnic_tribute_price
	uses_maintenance = no
	price_multiplier = {
		add = {
			desc = "TFE_HUNNIC_TRIBUTE_MULTIPLIER"
			every_international_organization_member = {
				add = monthly_income_trade_and_tax
			}
			multiply = 0.01
		}
		# Demand Greater Tribute (generic_actions/tfe_hunnic_storm.txt)
		if = {
			limit = { leader_country ?= { has_variable = tfe_greater_tribute } }
			multiply = {
				desc = "tfe_demand_greater_tribute"
				value = 2
			}
		}
	}
	proportion_for_payer = {
		add = {
			desc = "tfe_hunnic_tribute"
			value = monthly_income_trade_and_tax
		}
		# Win the Huns' Favour (generic_actions/tfe_hunnic_storm.txt)
		if = {
			limit = { has_variable = tfe_hunnic_favour }
			multiply = {
				desc = "tfe_win_the_huns_favour"
				value = 0.5
			}
		}
	}
	proportion_for_payee = {
		add = {
			desc = "tfe_hunnic_tribute"
			value = 1
		}
	}
}

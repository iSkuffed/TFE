	max_countries = 1
	priority = 10
	modifier = { monthly_prestige = 0.05 }
	can_bestow_trigger = { is_member_of_international_organization = scope:recipient }

	auto_rescind_trigger = { NOT = { is_member_of_international_organization = scope:recipient } }
	on_bestowed_effect = {
		scope:recipient = { set_leader_country = root }
	}
	map_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_LEADER_COLOR

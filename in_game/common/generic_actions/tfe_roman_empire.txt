# honorius-only: the Imperium Romanum's seat (tools/test_western_rome.py)
# TFE: the Imperium Romanum window's button for the West (Stilicho's Claims is a decision now,
# decisions/tfe_fall_of_the_west.txt). It costs Unity (the effects run as the West's actor).
tfe_align_visigoths = {
	type = internationalorganization
	icon = migrate_pop_based_country

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		exists = international_organization:tfe_roman_empire
		scope:actor = { tag = WRE }
		country_exists = c:VIS
	}
	allow = {
		scope:actor = { NOT = { has_variable = tfe_visigoths_aligned } }
	}

	select_trigger = {
		looking_for_a = international_organization
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = { international_organization_type = international_organization_type:tfe_roman_empire }
	}

	effect = {
		scope:actor = {
			set_variable = tfe_visigoths_aligned
			c:VIS ?= {
				if = {
					limit = { is_subject_type = tfe_foederati }
					overlord = { cancel_subject = c:VIS }
				}
			}
			create_relation = { first = scope:actor second = c:VIS type = relation_type:alliance }
		}
		custom_tooltip = tfe_unity_down_10_tt
		hidden_effect = { scope:recipient = { change_variable = { name = tfe_unity add = -10 } } }
	}
	ai_will_do = { value = -300 }
}

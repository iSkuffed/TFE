# TFE: migratory peoples (army-based hosts). A host may once give up its homeland for a great host that
# costs nothing while it owns no land (auto_modifiers/tfe_migratory.txt). It cannot replenish: landless, it
# has no manpower, so every warrior lost is gone until it wins land of its own.
tfe_start_migration = {
	type = owncountry
	icon = migrate_pop_based_country

	potential = {
		scope:actor = {
			country_type = army
			NOT = { has_variable = tfe_migrating }
		}
	}
	allow = {
		scope:actor = { any_army = { always = yes } }
	}

	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	effect = {
		custom_tooltip = tfe_start_migration_tt
		hidden_effect = {
			scope:actor = { tfe_start_migration_effect = yes }	# scripted_effects/tfe_migratory.txt
		}
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { scope:actor = { at_war = no } }
			add = 50
		}
	}
}

# TFE: Migrate into Rome, a Native Decision shown in the decisions tab (written by script/decisions.py)
tfe_migrate = {
	type = owncountry
	show_in_decision_panel = yes
	decision_category = tfe_fall_of_the_west
	potential = {
		situation:tfe_decline_of_the_west = { situation_is_active = yes }
		scope:actor = {
			tfe_is_migrator = yes
			NOT = { tag = SLF }	# the Salians raid over the Rhine instead (decisions/tfe_barbarian_kingdoms.txt)
			OR = {
				NOT = { has_variable = tfe_migrating }
				has_variable = tfe_settled	# a settled host may take the road again
			}
			NOT = { has_variable = tfe_host_kingdom }	# until it reforms into a monarchy (decisions/tfe_barbarian_kingdoms.txt)
		}
		any_country = { tfe_is_roman_state = yes }
	}
	allow = {
		scope:actor = {
			custom_tooltip = {
				text = tfe_migration_not_yet_tt
				current_date >= 395.2.18	# a month after the game starts (defines' START_DATE)
			}
			is_subject = no	# no warband afield is fine: the host gathers at the capital
			tfe_frontier_unmanned = yes	# scripted_triggers/tfe_decline_rome.txt: a manned frontier holds them
		}
	}
	ai_tick = monthly
	ai_tick_frequency = 1
	automation_tick = never
	automation_tick_frequency = 12
	# the Roman state to march on: either Empire, Stilicho's West, or a successor born of a revolt against one
	# (scripted_triggers/tfe_western_rome.txt); one that holds land, for the war to have a goal
	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			every_country = {
				limit = { tfe_is_roman_state = yes }
				add_to_list = source
			}
		}
		target_flag = target_rome
		name = "tfe_migrate_choose_rome"
		none_available_msg_key = "tfe_migrate_no_rome"
		column = {
			data = name
		}
		visible = {
			tfe_is_roman_state = yes
			any_owned_location = { always = yes }
		}
	}
	# the province to march on, the war's goal (casus_belli/tfe_migration.txt). The AI takes one on its own border
	# if the Rome it chose has any there, and never the capital's: a host bound for Constantinople never arrives.
	select_trigger = {
		looking_for_a = province
		interaction_source_list = {
			scope:target_rome = { every_province = { add_to_list = source } }
		}
		target_flag = target_province
		name = "tfe_migrate_choose_province"
		none_available_msg_key = "tfe_migrate_no_province"
		column = {
			data = name
		}
		visible = {
			OR = {
				scope:actor = { is_ai = no }
				AND = {
					NOT = { any_location_in_province = { is_capital = yes } }
					OR = {
						any_location_in_province = { any_neighbor_location = { owner ?= scope:actor } }
						NOT = {
							scope:target_rome = {
								any_province = {
									any_location_in_province = { any_neighbor_location = { owner ?= scope:actor } }
								}
							}
						}
					}
				}
			}
		}
	}
	effect = {
		custom_tooltip = tfe_start_migration_tt
		custom_tooltip = tfe_migrate_tt
		hidden_effect = {
			scope:actor = {
				tfe_start_migration_effect = yes	# scripted_effects/tfe_migratory.txt
				declare_war_with_cb = {
					target = scope:target_rome
					type = casus_belli:cb_tfe_migration
					target_province = scope:target_province
				}
			}
		}
	}
	# Pace the chaos: one AI people on the road at a time. None moves within a year of the last host setting out
	# (tfe_host_took_the_road), and then only when pushed: the Huns at the door or Rome divided. A settled AI host
	# stays settled. Of the Roman states, the one across the river first; a long trek only by chance.
	ai_will_do = {
		value = 0
		if = {
			limit = {
				NOT = { has_global_variable = tfe_host_took_the_road }
				scope:actor = {
					at_war = no
					NOT = { has_variable = tfe_settled }
				}
			}
			add = 5
			if = {
				limit = { scope:actor = { any_neighbor_country = { tfe_is_under_the_yoke = yes } } }
				add = 45
			}
			if = {
				limit = { scope:actor = { any_neighbor_country = { this = scope:target_rome } } }
				add = 10
			}
			if = {
				limit = { international_organization:tfe_roman_empire ?= { var:tfe_unity < 50 } }
				add = 25
			}
		}
	}
}

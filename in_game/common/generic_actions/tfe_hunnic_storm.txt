# TFE: the actions of the Hunnic Storm (situations/tfe_hunnic_storm.txt)
# A member of the Hunnic Yoke rises against the Huns (vanilla break_the_yoke), from the Hunnic Storm's panel
tfe_break_the_yoke = {
	type = situation
	force_click_and_confirm_or_hold = yes

	potential = {
		situation:tfe_hunnic_storm = { situation_is_active = yes }
		scope:actor = {
			is_member_of_international_organization = international_organization:tfe_hunnic_yoke
			NOT = {
				is_leader_of_international_organization = international_organization:tfe_hunnic_yoke
			}
		}
	}

	allow = {
		scope:actor = {
			is_subject = no
			NOT = { is_at_war_with = international_organization:tfe_hunnic_yoke.leader_country }
			NOT = { has_truce_with = international_organization:tfe_hunnic_yoke.leader_country }
		}
	}

	ai_tick = daily
	ai_tick_frequency = 80

	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = tfe_break_the_yoke
		years = 10
	}

	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:tfe_hunnic_storm = {
				add_to_list = source
			}
		}
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:tfe_hunnic_storm = this
			situation_is_active = yes
		}
	}

	effect = {
		if = {
			limit = {
				exists = international_organization:tfe_hunnic_yoke.leader_country
			}
			custom_tooltip = {
				text = tfe_break_the_yoke_tt
				scope:actor = {
					leave_all_wars_with = international_organization:tfe_hunnic_yoke.leader_country
					declare_war_with_cb = {
						target = international_organization:tfe_hunnic_yoke.leader_country
						type = casus_belli:cb_tfe_break_the_yoke
					}
				}
			}
		}
		else = { custom_tooltip = tfe_break_the_yoke_tt }
	}

	ai_will_do = {
		add = -20
		international_organization:tfe_hunnic_yoke.leader_country ?= {
			save_temporary_scope_as = target_country
		}
		add = {
			if = {
				limit = {
					exists = international_organization:tfe_hunnic_yoke.leader_country
					scope:actor = {
						total_population > scope:target_country.total_population
					}
				}
				add = 30
			}
		}
	}
}

# Phase 2: the ruler of the Huns takes the title Scourge of God, once per Storm. His death opens the Reckoning.
tfe_take_the_scourge_title = {
	type = situation

	potential = {
		scope:actor = { tag = HNS }
		situation:tfe_hunnic_storm = {
			situation_is_active = yes
			var:tfe_storm_phase = 2
			NOT = { has_variable = tfe_scourge }
		}
	}
	allow = {
		scope:actor = {
			ruler ?= { is_adult = yes }
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 1
	automation_tick = never
	automation_tick_frequency = 12

	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:tfe_hunnic_storm = {
				add_to_list = source
			}
		}
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:tfe_hunnic_storm = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor.ruler = {
			add_trait = trait:tfe_scourge_of_god
			save_scope_as = tfe_new_scourge
		}
		custom_tooltip = tfe_take_the_scourge_title_tt
		hidden_effect = {
			situation:tfe_hunnic_storm = {
				set_variable = { name = tfe_scourge value = scope:tfe_new_scourge }
			}
		}
	}

	ai_will_do = {
		add = 1000
	}
}

# The Huns: a great feast for the kings of the Yoke. Fed kings are slower to rise in the Reckoning.
tfe_feast_the_subject_kings = {
	type = situation

	potential = {
		scope:actor = { tag = HNS }
		situation:tfe_hunnic_storm = { situation_is_active = yes }
	}
	allow = {
		scope:actor = {
			at_war = no
		}
	}

	price = price:tfe_hunnic_storm_gold
	price_modifier = {
		add = {
			desc = "tfe_one_years_income"
			value = scope:actor.monthly_income_trade_and_tax
			multiply = 12
			min = 50
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = tfe_feast_the_subject_kings
		years = 10
	}

	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:tfe_hunnic_storm = {
				add_to_list = source
			}
		}
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:tfe_hunnic_storm = this
			situation_is_active = yes
		}
	}

	effect = {
		custom_tooltip = tfe_feast_the_subject_kings_tt
		every_country = {
			limit = {
				tfe_is_under_the_yoke = yes
				NOT = { tag = HNS }
			}
			add_opinion = { target = c:HNS modifier = tfe_hunnic_feast }
			hidden_effect = { set_variable = { name = tfe_feasted_by_the_huns value = yes years = 10 } }
		}
	}

	ai_will_do = {
		add = 0
		# before and during the Scourge, while the Reckoning is still to come
		if = {
			limit = {
				situation:tfe_hunnic_storm = { var:tfe_storm_phase = 2 }
			}
			add = 50
		}
	}
}

			limit = {

# A Roman empire hires Hunnic riders as mercenaries, as Aetius did

	type = situation

	potential = {
		situation:tfe_hunnic_storm = { situation_is_active = yes }
		country_exists = c:HNS
		scope:actor = {
			OR = {
				tfe_is_western_rome = yes
				tag = EAR
			}
		}
	}
	allow = {
		scope:actor = {
			NOT = { is_at_war_with = c:HNS }
		}
	}

	price = price:tfe_hunnic_storm_gold
	price_modifier = {
		add = {
			desc = "tfe_three_months_income"
			value = scope:actor.monthly_income_trade_and_tax
			multiply = 3
			min = 25
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = tfe_hire_hunnic_horsemen
		years = 10
	}

	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:tfe_hunnic_storm = {
				add_to_list = source
			}
		}
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:tfe_hunnic_storm = this
			situation_is_active = yes
		}
	}

	effect = {
		custom_tooltip = tfe_hire_hunnic_horsemen_tt
		hidden_effect = {
			create_mercenary = {
				name = tfe_hunnic_horsemen
				home = scope:actor.capital
				categories = {
					army_light_cavalry = 6
				}
				save_scope_as = tfe_hunnic_horsemen
				months = 120
			}
			if = {
				limit = { exists = scope:tfe_hunnic_horsemen }
				scope:actor = { hire_mercenary = scope:tfe_hunnic_horsemen }
			}
		}
	}

	ai_will_do = {
		add = 0
		if = {
			limit = {
				scope:actor = { at_war = yes }
			}
			add = 20
		}
	}

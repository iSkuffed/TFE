namespace = tfe_walls

# 1 Jan 405 (on_action/tfe_opening.txt): the great land walls are offered to Constantinople's owner, eight years
# before the historical date. in_game/common/building_types/tfe_theodosian_walls.txt keeps them out of the
# buildings tab until then. Accepting starts construction at a third off the tab's price; refusing leaves the tab's price.
tfe_walls.1 = {
	type = country_event
	title = tfe_walls.1.title
	desc = tfe_walls.1.desc
	outcome = positive
	image = "gfx/interface/illustrations/event/backgrounds/exterior/byz_burghers_exterior.dds"
	trigger = {
		owns = location:constantinople
		location:constantinople = { NOT = { has_building = building_type:theodosian_walls } }
	}
	# start the walls now at a third off the tab's price
	option = {
		name = tfe_walls.1.a
		location:constantinople = {
			construct_building = {
				building_type = building_type:theodosian_walls
				cost_multiplier = 0.67
				cost_multiplier_reason = "The wall levy"
			}
		}
		add_prestige = 10
		ai_chance = { base = 3 }
	}
	# they stay in the buildings tab at full price
	option = {
		name = tfe_walls.1.b
		ai_chance = { base = 1 }
	}
}

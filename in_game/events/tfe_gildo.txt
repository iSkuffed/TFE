namespace = tfe_gildo

# Gildo's revolt is earned, not dated (on_action/tfe_gildo.txt starts the count). Every month the West's
# tfe_gildo_progress rises by tfe_gildo_monthly_rise; three warnings offer ways to slow it; at 100, or after
# 14 August 397, he rises (5, then tfe_opening.3 for the split). tfe_gildo_revolt_fires reads the result.
# the monthly count (hidden)
tfe_gildo.1 = {
	type = country_event
	title = tfe_gildo.1.title
	desc = tfe_gildo.1.desc
	outcome = neutral
	hidden = yes
	trigger = { NOT = { has_variable = tfe_gildo_revolt_fired } }
	immediate = {
		change_variable = { name = tfe_gildo_progress add = tfe_gildo_monthly_rise }
		if = {
			limit = { var:tfe_gildo_progress < 0 }
			set_variable = { name = tfe_gildo_progress value = 0 }
		}
		if = {
			limit = {
				OR = {
					var:tfe_gildo_progress >= 100
					current_date > 397.8.14
				}
			}
			trigger_event_non_silently = tfe_gildo.5
		}
		else = {
			if = {
				limit = { var:tfe_gildo_progress >= 35 NOT = { has_variable = tfe_gildo_warned_1 } }
				set_variable = tfe_gildo_warned_1
				trigger_event_non_silently = tfe_gildo.2
			}
			if = {
				limit = { var:tfe_gildo_progress >= 65 NOT = { has_variable = tfe_gildo_warned_2 } }
				set_variable = tfe_gildo_warned_2
				trigger_event_non_silently = tfe_gildo.3
			}
			if = {
				limit = { var:tfe_gildo_progress >= 90 NOT = { has_variable = tfe_gildo_warned_3 } }
				set_variable = tfe_gildo_warned_3
				trigger_event_non_silently = tfe_gildo.4
			}
			trigger_event_silently = { id = tfe_gildo.1 months = 1 }
		}
	}
}

# after the rising, monthly for two years: the East's homage once his war is over, and his seat (hidden).
# He rises with all of Africa but Tingitana (scripted_triggers/tfe_gildo.txt), so no town is left to win.
tfe_gildo.6 = {
	type = country_event
	title = tfe_gildo.6.title
	desc = tfe_gildo.6.desc
	outcome = neutral
	hidden = yes
	trigger = { country_exists = c:GILDO }
	immediate = {
		change_variable = { name = tfe_gildo_wave add = 1 }
		# the East took his oath (tfe_opening.4): once his war with us is over, he is Constantinople's man
		if = {
			limit = { has_variable = tfe_gildo_goes_east c:GILDO = { NOT = { is_at_war_with = root } } }
			c:GILDO = {
				make_subject_of = {
					target = c:EAR
					type = subject_type:vassal
				}
			}
			remove_variable = tfe_gildo_goes_east
			# the West is told the oath is sealed (7); Roman Unity falls as it is told
			hidden_effect = {
				international_organization:tfe_roman_empire ?= {
					change_variable = { name = tfe_unity add = -10 }
				}
			}
			trigger_event_non_silently = tfe_gildo.7
		}
		# Carthage decides where he sits
		if = {
			limit = { location:tunis = { owner ?= c:GILDO } }
			c:GILDO = { set_capital = location:tunis }
		}
		if = {
			limit = { var:tfe_gildo_wave < 24 }
			trigger_event_silently = { id = tfe_gildo.6 months = 1 }
		}
	}
}

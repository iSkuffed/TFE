namespace = tfe_barbarian_kingdoms

# A host has settled (on_action/tfe_migratory.txt fires this a day later, once the whole peace or Hospitalitas
# has handed it its land). Its people (tfe_host_people) spread evenly over every location it owns, and their
# number grows with the land: all of them on 5 locations, a share on fewer, and up to 2x on
# 10 or more. So each location gets people / 5, or people x 2 / locations past that.
tfe_barbarian_kingdoms.1 = {
	type = country_event
	title = tfe_barbarian_kingdoms.1.title
	hidden = yes
	trigger = { has_variable = tfe_host_people }
	immediate = {
		set_variable = { name = tfe_host_share value = var:tfe_host_people }
		if = {
			limit = { num_locations >= 10 }
			change_variable = { name = tfe_host_share multiply = 2 }
			change_variable = { name = tfe_host_share divide = num_locations }
		}
		else = {
			change_variable = { name = tfe_host_share divide = 5 }
		}
		every_owned_location = {
			add_pop = { culture = root.culture religion = root.religion type = pop_type:peasants size = root.var:tfe_host_share }
			# land the host's own people settled is quick to integrate: no clock to race, conquests after the
			# migration ends integrate at vanilla's pace
			add_location_modifier = { modifier = tfe_settled_by_the_host }
		}
		remove_variable = tfe_host_people
		remove_variable = tfe_host_share
	}
}

# A host has reformed into a monarchy (tfe_reform_into_a_monarchy, a day before): its nobles get the Auxilium et
# Consilium the start grant could not give a people that was no monarchy. Vanilla's privilege also needs the
# Knights advance, rechecked whenever the faith changes, so the crown brings feudalism and knights with it.
tfe_barbarian_kingdoms.2 = {
	type = country_event
	title = tfe_barbarian_kingdoms.2.title
	hidden = yes
	immediate = {
		if = {
			limit = { NOT = { has_advance = feudalism_advance } }
			research_advance = advance_type:feudalism_advance
		}
		if = {
			limit = { NOT = { has_advance = noble_knights } }
			research_advance = advance_type:noble_knights
		}
		if = {
			limit = { NOT = { has_estate_privilege = estate_privilege:auxilium_et_consilium } }
			grant_estate_privilege = estate_privilege:auxilium_et_consilium
		}
	}
}

# A host on the road's war has ended (on_action/tfe_barbarian_kingdoms.txt fires this a day later, once the peace
# has handed over its land). A host that won land settled with it (on_action/tfe_migratory.txt). One that won its
# Migrate into Rome war but took gold instead gets the province it marched on (tfe_promised_province) all the same.
# One that lost, or made peace with nothing, breaks up.
tfe_barbarian_kingdoms.3 = {
	type = country_event
	title = tfe_barbarian_kingdoms.3.title
	hidden = yes
	trigger = {
		has_variable = tfe_migrating
		NOT = { has_variable = tfe_settled }
		NOT = { any_owned_location = { always = yes } }
	}
	immediate = {
		if = {
			limit = { has_variable = tfe_promised_province }
			var:tfe_promised_province = {
				every_location_in_province = { change_location_owner = root }
			}
		}
		else = {
			trigger_event_non_silently = tfe_barbarian_kingdoms.4
		}
	}
}

# The host breaks up: its warriors scatter, and the people who followed it go home to the remnant of the land it
# left (tfe_remnant_of, scripted_effects/tfe_migratory.txt). A host that left no remnant is simply gone.
tfe_barbarian_kingdoms.4 = {
	type = country_event
	title = tfe_barbarian_kingdoms.4.title
	desc = tfe_barbarian_kingdoms.4.desc
	outcome = negative
	image = "gfx/interface/illustrations/event/backgrounds/exterior/soldiers/north_german_soldiers_exterior.dds"
	option = {
		name = tfe_barbarian_kingdoms.4.a
		custom_tooltip = tfe_host_scatters_tt
		hidden_effect = {
			random_country = {
				limit = { var:tfe_remnant_of ?= root }
				capital ?= {
					add_pop = { culture = root.culture religion = root.religion type = pop_type:peasants size = root.var:tfe_host_people }
				}
			}
			every_army = { destroy_unit = yes }
			change_country_type = location	# landless and armyless: the country is gone
		}
	}
}

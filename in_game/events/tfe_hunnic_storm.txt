# The Huns press on the Germanic hosts: take to the road, or stay and risk the Yoke

	type = country_event
	category = situation_event
	title = tfe_hunnic_storm.2.title
	desc = tfe_hunnic_storm.2.desc
	outcome = neutral
	image = "gfx/interface/illustrations/situation/tfe_hunnic_storm.dds"

	trigger = {
		tfe_is_migrator = yes	# scripted_triggers/tfe_decline_of_the_west.txt
		NOT = { has_variable = tfe_migrating }
		is_subject = no
		tfe_barred_by_the_limes = no	# scripted_triggers/tfe_decline_rome.txt
	}

	# the road west (the Decline of the West's Migrate, generic_actions/tfe_migratory.txt)
	option = {
		name = tfe_hunnic_storm.2.a
		historical_option = yes
		custom_tooltip = tfe_start_migration_tt
		hidden_effect = { tfe_start_migration_effect = yes }	# scripted_effects/tfe_migratory.txt
		ai_chance = {
			base = 3
			modifier = {	# pace the chaos: one host on the road every 4 years
				factor = 0
				has_global_variable = tfe_host_took_the_road
			}
		}
	}

	option = {
		name = tfe_hunnic_storm.2.b
		add_prestige = 5
		ai_chance = { base = 1 }
	}
}

# The Reckoning: the Scourge is dead, and every people under the Yoke may rise now
tfe_hunnic_storm.3 = {
	type = country_event
	category = situation_event
	title = tfe_hunnic_storm.3.title
	desc = tfe_hunnic_storm.3.desc
	outcome = positive
	image = "gfx/interface/illustrations/situation/tfe_hunnic_storm.dds"

	trigger = {
		tfe_is_under_the_yoke = yes
		NOT = { tag = HNS }
		country_exists = c:HNS
	}

	# rise with the others
	option = {
		name = tfe_hunnic_storm.3.a
		trigger = {
			OR = {
				is_subject = no
				is_subject_type = tributary
			}
			NOT = { is_at_war_with = c:HNS }
		}
		# a tributary of the Huns just stops paying; a free member goes to war
		custom_tooltip = {
			text = tfe_break_the_yoke_tt
			trigger = { is_subject = no }
		}
		custom_tooltip = {
			text = tfe_cast_off_the_tribute_tt
			trigger = { is_subject_type = tributary }
		}
		hidden_effect = {
			if = {
				limit = { is_subject_type = tributary }
				c:HNS = { cancel_subject = root }
			}
			else = {
				leave_all_wars_with = c:HNS
				declare_war_with_cb = {
					target = c:HNS
					type = casus_belli:cb_tfe_break_the_yoke
				}
			}
		}
		ai_chance = {
			base = 1
			# the Huns are weak: no grown heir to hold the Yoke, or a smaller army than ours
			modifier = {
				factor = 3
				c:HNS = {
					NOT = { heir ?= { is_adult = yes } }
				}
			}
			modifier = {
				factor = 5
				c:HNS = { military_strength < root.military_strength }
			}
			# Feast the Subject Kings and Demand Greater Tribute (generic_actions/tfe_hunnic_storm.txt)
			modifier = {
				factor = 0.3
				has_variable = tfe_feasted_by_the_huns
			}
			modifier = {
				factor = 2
				has_variable = tfe_bled_by_the_huns
			}
		}
	}

	# stay loyal
	option = {
		name = tfe_hunnic_storm.3.b
		ai_chance = { base = 2 }
	}
}

# 395: Stilicho says the dying Theodosius gave him both sons to guard (on_action/tfe_opening.txt, game start)
tfe_opening.1 = {
	type = country_event
	title = tfe_opening.1.title
	desc = tfe_opening.1.desc
	outcome = neutral
	image = "gfx/interface/illustrations/event/backgrounds/exterior/byz_soldiers_exterior.dds"
	trigger = { character:tfe_stilicho ?= { is_alive = yes } }
	# send the Eastern legions home under Gainas
	option = {
		name = tfe_opening.1.a
		historical_option = yes
		set_variable = tfe_gainas_returned	# read by 2: the Goths are already in Constantinople
		custom_tooltip = tfe_unity_up_5_tt
		custom_tooltip = tfe_opening.1.a.tt
		hidden_effect = {
			international_organization:tfe_roman_empire ?= {
				change_variable = { name = tfe_unity add = 5 }
			}
		}
		ai_chance = { base = 3 }
	}
	# the guardianship of the East too: march on Constantinople's prefect
	option = {
		name = tfe_opening.1.b
		set_variable = tfe_stilicho_claims_the_east	# read by 2 and 4: the East remembers
		# the Eastern legions stay in Italy, free of any recruitment cost
		location:milano = {
			create_num_sub_unit_of_category = { count = 4 category = sub_unit_category:army_heavy_infantry }
			create_num_sub_unit_of_category = { count = 2 category = sub_unit_category:army_heavy_cavalry }
		}
		custom_tooltip = tfe_unity_down_15_tt
		hidden_effect = {
			international_organization:tfe_roman_empire ?= {
				change_variable = { name = tfe_unity add = -15 }
			}
		}
		# and the East, whose army this was, never forgives it (script/opening_events.py biases())
		custom_tooltip = tfe_legions_kept_tt
		hidden_effect = {
			c:EAR ?= { add_opinion = { target = c:WRE modifier = tfe_legions_kept } }
		}
		ai_chance = { base = 1 }
	}
}

# 27 November 395: Gainas's Goths cut Rufinus down on the Campus Martius at Constantinople
tfe_opening.2 = {
	type = country_event
	title = tfe_opening.2.title
	desc = tfe_opening.2.desc
	outcome = neutral
	image = "gfx/interface/illustrations/event/backgrounds/exterior/byz_soldiers_exterior.dds"
	trigger = { character:tfe_rufinus ?= { is_alive = yes } }
	immediate = {
		kill_character = character:tfe_rufinus
		remove_country_modifier = tfe_rufinus_prefecture
		# Stilicho's hand is seen in it
		if = {
			limit = { c:WRE ?= { has_variable = tfe_stilicho_claims_the_east } }

			hidden_effect = {
				international_organization:tfe_roman_empire ?= {
					change_variable = { name = tfe_unity add = -5 }
				}
			}
		}
		# Gainas's Goths were already inside the walls
		if = {
			limit = { c:WRE ?= { has_variable = tfe_gainas_returned } }
			custom_tooltip = tfe_unity_up_5_tt
			hidden_effect = {
				international_organization:tfe_roman_empire ?= {
					change_variable = { name = tfe_unity add = 5 }
				}
			}
		}
	}
	# the chamberlain holds the Augustus now
	option = {
		name = tfe_opening.2.a
		historical_option = yes
		trigger = { character:tfe_eutropius ?= { is_alive = yes } }
		add_country_modifier = { modifier = tfe_eutropius_chamber years = 4 }	# to his fall in 399
		ai_chance = { base = 3 }
	}
	# Arcadius will rule for himself
	option = {
		name = tfe_opening.2.b
		add_legitimacy = 10
		add_stability = -10
		ai_chance = { base = 1 }
	}
}

# Gildo, count of Africa, holds back Rome's grain and offers Africa to the East (fired by tfe_gildo.5, autumn 397 at the latest)
tfe_opening.3 = {
	type = country_event
	title = tfe_opening.3.title
	desc = tfe_opening.3.desc
	outcome = negative
	image = "gfx/interface/illustrations/event/backgrounds/exterior/nobles/syrian_nobles_exterior.dds"
	trigger = {
		OR = {
			location:cherchell = { owner ?= root }
			location:constantine_ALG = { owner ?= root }
		}
	}
	immediate = {
		# a revolt, not a war: the war screen can then annex the revolter outright, with no antagonism.
		# The revolter appears at once, named after its biggest town, and tfe_opening.7 makes it Gildo's kingdom in the
		# same moment: a day's wait showed the revolt's patchwork of rebel countries for that day (user)
		create_rebel = {
			category = nationalist
			name = tfe_gildo_rebels
			religion = religion:donatism
			culture = culture:afro_roman
			save_scope_as = tfe_gildo_rebels
		}
		every_owned_location = {
			limit = { tfe_gildo_base_land = yes }
			every_pop = { change_pop_allegiance = scope:tfe_gildo_rebels }
		}
		scope:tfe_gildo_rebels ?= { start_revolt = yes }
		trigger_event_silently = tfe_opening.7
	}
	# his brother Mascezel knows Africa and hates him: send him
	option = {
		name = tfe_opening.3.a
		historical_option = yes
		ai_chance = { base = 1 }
	}
}

# The moment of the rising: the revolter becomes Gildo's kingdom and takes the rest of his land (fired by 3)
tfe_opening.7 = {
	type = country_event
	title = tfe_opening.7.title
	hidden = yes
	immediate = {
		# risen with all Africa the revolt splits into more than one rebel country: crown the war's leader
		every_current_war = {
			limit = {
				attacker_leader = {
					is_at_war_with = root
					any_owned_location = { tfe_gildo_base_land = yes }
					# carved from his land alone: the Mauri who back him sit in Laghouat too, but hold land beyond it
					NOT = { any_owned_location = { NOT = { tfe_gildo_base_land = yes } } }
				}
			}
			attacker_leader = {
				save_scope_as = tfe_usurper
			}
		}
		if = {
			limit = { NOT = { exists = scope:tfe_usurper } }
			# the Mauri lead the rebel side as backers: the revolter is the one holding nothing else
			random_country = {
				limit = {
					is_at_war_with = root
					any_owned_location = { tfe_gildo_base_land = yes }
					# carved from his land alone: the Mauri who back him sit in Laghouat too, but hold land beyond it
					NOT = { any_owned_location = { NOT = { tfe_gildo_base_land = yes } } }
				}
				save_scope_as = tfe_usurper
			}
		}
		# Gildo, or if he has died before his hold ripens, his Moorish kinsman in his place
		if = {
			limit = { character:tfe_gildo ?= { is_alive = yes } }
			character:tfe_gildo = { save_scope_as = tfe_usurper_ruler }
		}
		scope:tfe_usurper ?= {
			define_unique_country_tag = GILDO
			change_country_name = TFE_GILDO
			change_country_adjective = TFE_GILDO_ADJ
			change_country_color = map_gutnish
			change_country_flag = TFE_GILDO
			set_country_rank = country_rank:rank_kingdom
			change_government_type = government_type:monarchy
			add_gold = 200	# Africa's taxes, kept back from Italy
			if = {
				limit = { NOT = { exists = scope:tfe_usurper_ruler } }
				create_character = {
					first_name = name_gildo
					age = 35
					religion = religion:donatism
					estate = estate_type:nobles_estate
					culture = culture:afro_roman
					mil = 60
					save_scope_as = tfe_usurper_ruler
				}
			}
			set_new_ruler = scope:tfe_usurper_ruler
			set_variable = { name = tfe_usurper_against value = root years = 10 }
			# a neighbour backing the revolt (the Mauri) leads the rebel side, and Annex Revolter then only offers it
			# a white peace: send the backers home so the war screen can annex Gildo. Only backers, which hold land
			# beyond Africa: sending home the other rebel country the revolt split off ends the whole war (user, in game)
			every_current_war = {
				limit = { is_in_war = root }
				save_scope_as = tfe_gildo_war
				every_war_participant = {
					limit = {
						is_at_war_with = root
						NOT = { this = scope:tfe_usurper }
						any_owned_location = { NOT = { tfe_gildo_base_land = yes } }
					}
					leave_war = {
						war = scope:tfe_gildo_war
						actor = root
					}
				}
			}
			# the revolt system makes the revolter a Secessionist subject of its Kabyle backers (the Mauri); Gildo is
			# his own master, not the Mauri's
			if = {
				limit = { is_subject = yes }
				overlord = { cancel_subject = prev }
			}
		}
		if = {
			limit = { exists = scope:tfe_usurper }
			# the revolt hands him only part of it: the rest of Africa but Tingitana is his by right
			every_owned_location = {
				limit = { tfe_gildo_base_land = yes }
				change_location_owner = scope:tfe_usurper
			}
			# his seat is Carthage, which he takes with the rest of Africa; Cherchell or Cirta if the West holds it
			if = {
				limit = { location:tunis = { owner ?= scope:tfe_usurper } }
				scope:tfe_usurper = { set_capital = location:tunis }
			}
			else_if = {
				limit = { location:cherchell = { owner ?= scope:tfe_usurper } }
				scope:tfe_usurper = { set_capital = location:cherchell }
			}
			else_if = {
				limit = { location:constantine_ALG = { owner ?= scope:tfe_usurper } }
				scope:tfe_usurper = { set_capital = location:constantine_ALG }
			}
			set_variable = { name = tfe_usurper value = scope:tfe_usurper years = 10 }
			set_variable = { name = tfe_gildo_wave value = 0 }
			trigger_event_silently = { id = tfe_gildo.6 months = 3 }	# the East's homage and his seat, monthly
			c:EAR ?= { trigger_event_non_silently = tfe_opening.4 }
		}
	}
}

# The East is offered the diocese of Africa (fired by 7)
tfe_opening.4 = {
	type = country_event
	title = tfe_opening.4.title
	desc = tfe_opening.4.desc
	outcome = neutral
	image = "gfx/interface/illustrations/event/backgrounds/interior/byz_nobles_interior.dds"
	trigger = { c:WRE ?= { has_variable = tfe_usurper } }
	immediate = {
		c:WRE.var:tfe_usurper = { save_scope_as = tfe_usurper }
	}
	# accept it: Africa's grain now sails for Constantinople
	option = {
		name = tfe_opening.4.a
		historical_option = yes
		# not now: a vassal of the East cannot be annexed as a revolter, and the revolt war ends (tfe_gildo.6)
		custom_tooltip = tfe_gildo_goes_east_tt
		hidden_effect = { c:WRE = { set_variable = tfe_gildo_goes_east } }
		custom_tooltip = tfe_unity_down_15_tt
		hidden_effect = {
			international_organization:tfe_roman_empire ?= {
				change_variable = { name = tfe_unity add = -15 }
			}
		}
		# the West is told the same day: tfe_opening.8
		hidden_effect = { c:WRE ?= { trigger_event_non_silently = tfe_opening.8 } }
		ai_chance = {
			base = 10
			# Constantinople takes Africa unless it loves or fears the West: friendship and a united Empire hold it
			# back, a rival, a broken Empire and Stilicho's hand on the Eastern legions push it forward
			modifier = {
				factor = 0.2
				opinion = {
					target = c:WRE
					value >= 50
				}
			}
			modifier = {
				factor = 0.5
				trust = {
					target = c:WRE
					value >= 50
				}
			}
			modifier = {
				factor = 0.3
				international_organization:tfe_roman_empire ?= { var:tfe_unity >= 60 }
			}
			modifier = {
				factor = 3
				is_rival_of = c:WRE
			}
			modifier = {
				factor = 2
				international_organization:tfe_roman_empire ?= { var:tfe_unity < 30 }
			}
			modifier = {
				factor = 3
				c:WRE ?= { has_variable = tfe_stilicho_claims_the_east }
			}
		}
	}
	# one empire: send his envoys back to Carthage in chains
	option = {
		name = tfe_opening.4.b
		custom_tooltip = tfe_unity_up_5_tt
		hidden_effect = {
			international_organization:tfe_roman_empire ?= {
				change_variable = { name = tfe_unity add = 5 }
			}
		}
		ai_chance = { base = 10 }
	}
}

# 400: Honorius comes of age. His heir by blood is Arcadius, and a foreign ruler as heir makes the halves a union:
# Stilicho's son Eucherius instead, suspected of being groomed for the purple, betrothed to Galla Placidia
tfe_opening.6 = {
	type = country_event
	title = tfe_opening.6.title
	hidden = yes
	fire_only_once = yes
	trigger = {
		ruler ?= character:tfe_honorius
		character:tfe_eucherius ?= {
			is_alive = yes
			employer ?= root
		}
	}
	immediate = {
		set_as_designated_heir = character:tfe_eucherius
	}
}

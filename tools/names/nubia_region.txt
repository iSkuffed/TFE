upper_butana_province = the Astaboras  # p: Astaboras, the Atbara river country (Strabo, Ptolemy)
upper_gezira_province = the Astapus  # p: Astapus, the Blue Nile (Eratosthenes via Strabo, Ptolemy)

darfur_area = Inner Ethiopia  # p: Aethiopia interior (Ptolemy 4.8), the unknown lands west and south of Meroe

south_darfur_province = the Western Savanna  # d: grasslands of southern Darfur, nothing ancient

equatoria_province = the Mountains of the Moon  # p: Lunae Montes (Ptolemy 4.8), the far south where the Nile rises

nubia_proper_area = Nobadia  # a: land of the Nobades/Nobatae (Silko inscription at Talmis, Procopius), Lower and Middle Nubia

makuria_province = Napata  # a: Napata (Strabo, Pliny, Ptolemy), old royal city at Jebel Barkal, sacked by Petronius 24 BC
nobatia_province = the Triacontaschoenus  # a: Triacontaschoenus (Gallus' Philae stele), Lower Nubia above the Dodecaschoenus; Talmis, Primis, held by Blemmyes and Nobatae
